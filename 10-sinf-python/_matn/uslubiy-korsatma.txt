# src/apps/news/models.py
from django.db import models
from django.utils.text import slugify

from apps.common.models import TimeStampedModel


class Category(TimeStampedModel):
    name = models.CharField(max_length=120, unique=True)
    slug = models.SlugField(max_length=140, unique=True)

    class Meta:
        verbose_name_plural = "Categories"
        ordering = ("name",)

    def save(self, *args, **kwargs):
        if not self.slug:
            self.slug = slugify(self.name)
        super().save(*args, **kwargs)

    def __str__(self) -> str:
        return self.name


class Tag(TimeStampedModel):
    name = models.CharField(max_length=80, unique=True)
    slug = models.SlugField(max_length=100, unique=True)

    class Meta:
        ordering = ("name",)

    def save(self, *args, **kwargs):
        if not self.slug:
            self.slug = slugify(self.name)
        super().save(*args, **kwargs)

    def __str__(self) -> str:
        return self.name


class PostStatus(models.TextChoices):
    DRAFT = "draft", "Draft"
    PUBLISHED = "published", "Published"
    ARCHIVED = "archived", "Archived"


class Post(TimeStampedModel):
    title = models.CharField(max_length=220)
    slug = models.SlugField(max_length=240, unique=True)
    excerpt = models.CharField(max_length=300, blank=True)
    body = models.TextField()

    cover_image = models.ImageField(upload_to="covers/", blank=True, null=True)

    status = models.CharField(
        max_length=12,
        choices=PostStatus.choices,
        default=PostStatus.DRAFT,
    )
    published_at = models.DateTimeField(blank=True, null=True)

    category = models.ForeignKey(
        Category,
        on_delete=models.PROTECT,
        related_name="posts",
    )
    tags = models.ManyToManyField(
        Tag,
        blank=True,
        related_name="posts",
    )

    class Meta:
        ordering = ("-published_at", "-created_at")
        indexes = [
            models.Index(fields=["slug"]),
            models.Index(fields=["status", "published_at"]),
        ]

    def save(self, *args, **kwargs):
        if not self.slug:
            self.slug = slugify(self.title)
        super().save(*args, **kwargs)

    def __str__(self) -> str:
        return self.title

Ushbu modul bo‘yicha izohlar:
	•	slug auto to‘ldiriladi (save() ichida).
	•	category — PROTECT: kategoriya o‘chsa postlar “yetim” bo‘lib qolmasin.
	•	tags — ManyToMany, ixtiyoriy.
	•	indexes — keyin search/filter tezroq bo‘lishi uchun.

# src/apps/comments/models.py
from django.db import models

from apps.common.models import TimeStampedModel
from apps.news.models import Post


class Comment(TimeStampedModel):
    post = models.ForeignKey(
        Post,
        on_delete=models.CASCADE,
        related_name="comments",
    )

from django.contrib.auth.base_user import AbstractBaseUser, BaseUserManager
from django.contrib.auth.models import PermissionsMixin
from django.db import models

from apps.common.models import TimeStampedModel


class UserManager(BaseUserManager):
    def create_user(self, email: str, password: str | None = None, **extra_fields):
        if not email:
            raise ValueError("Email is required")

        email = self.normalize_email(email)
        user = self.model(email=email, **extra_fields)
        user.set_password(password)
        user.save(using=self._db)
        return user

    def create_superuser(self, email: str, password: str, **extra_fields):
        extra_fields.setdefault("is_staff", True)
        extra_fields.setdefault("is_superuser", True)
        extra_fields.setdefault("is_active", True)

        if extra_fields.get("is_staff") is not True:
            raise ValueError("Superuser must have is_staff=True")
        if extra_fields.get("is_superuser") is not True:
            raise ValueError("Superuser must have is_superuser=True")

        return self.create_user(email=email, password=password, **extra_fields)


class User(TimeStampedModel, AbstractBaseUser, PermissionsMixin):
    email = models.EmailField(unique=True)
    full_name = models.CharField(max_length=160, blank=True)
    avatar = models.ImageField(upload_to="avatars/", blank=True, null=True)

    is_active = models.BooleanField(default=True)
    is_staff = models.BooleanField(default=False)

    objects = UserManager()

    USERNAME_FIELD = "email"
    REQUIRED_FIELDS: list[str] = []

    class Meta:
        ordering = ("-created_at",)

    def __str__(self) -> str:
        return self.email

    email = serializers.EmailField()
    password = serializers.CharField(write_only=True)

    def validate(self, attrs):
        user = authenticate(email=attrs["email"], password=attrs["password"])
        if not user:
            raise serializers.ValidationError("Invalid email or password")
        attrs["user"] = user
        return attrs

# src/apps/accounts/views.py
from django.contrib.auth import login as session_login
from rest_framework import generics, status
from rest_framework.authtoken.models import Token
from rest_framework.response import Response
from rest_framework.views import APIView

from apps.accounts.serializers import (
    RegisterSerializer,
    TokenLoginSerializer,
    UserSerializer,
)


class RegisterView(generics.CreateAPIView):
    serializer_class = RegisterSerializer


class TokenLoginView(APIView):
    """
    Email+password -> DRF Token qaytaradi.
    (Session login ham qilamiz, shunda browsable API’da qulay bo‘ladi)
    """

    def post(self, request):
        serializer = TokenLoginSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        user = serializer.validated_data["user"]

        # Session (cookie) login
        session_login(request, user)

        # Token yaratish / olish
        token, _ = Token.objects.get_or_create(user=user)

        return Response(
            {
                "token": token.key,
                "user": UserSerializer(user).data,
            },
            status=status.HTTP_200_OK,
        )


class MeView(APIView):
    def get(self, request):
        return Response(
            {
                "is_authenticated": bool(getattr(request.user, "is_authenticated", False)),
                "user": UserSerializer(request.user).data if request.user.is_authenticated else None,
            }
        )

3.1. apps/common/permissions.py
# src/apps/common/permissions.py
from rest_framework.permissions import SAFE_METHODS, BasePermission


class IsStaffOrReadOnly(BasePermission):
    """
    GET/HEAD/OPTIONS -> hammaga ruxsat
    POST/PUT/PATCH/DELETE -> faqat staff
    """

    def has_permission(self, request, view):
        if request.method in SAFE_METHODS:
            return True
        return bool(request.user and request.user.is_authenticated and request.user.is_staff)


class IsOwnerOrStaff(BasePermission):
    """
    Object-level: obyekt egasi yoki staff bo‘lsa ruxsat.
    Egasi bo‘ladigan field nomi default: 'author' yoki 'user'.
    ViewSet ichida owner_field = "author" yoki "user" deb berib ketamiz.
    """

    owner_field = None  # "author" yoki "user"

    def has_object_permission(self, request, view, obj):
        if request.method in SAFE_METHODS:
            return True

        user = request.user
        if not (user and user.is_authenticated):
            return False

        if user.is_staff:
            return True

        owner_field = getattr(view, "owner_field", None) or self.owner_field
        if not owner_field:
            return False

        owner = getattr(obj, owner_field, None)
        return owner == user

# src/apps/news/filters.py
import django_filters

from apps.news.models import Post


class PostFilter(django_filters.FilterSet):
    """
    Post filterlari:
    - status
    - category (id)
    - author (id)
    - tags (id) -> ko‘p teg: tags=1&tags=2 ishlaydi
    - published_at oralig‘i: published_from, published_to
    """

    published_from = django_filters.IsoDateTimeFilter(
        field_name="published_at",
        lookup_expr="gte",
    )
    published_to = django_filters.IsoDateTimeFilter(
        field_name="published_at",
        lookup_expr="lte",
    )

    class Meta:
        model = Post
        fields = {
            "status": ["exact"],
            "category": ["exact"],
            "author": ["exact"],
            "tags": ["exact"],
        }
Izoh:

from django.http import FileResponse, Http404


class PostViewSet(viewsets.ModelViewSet):
    # ...

    @action(methods=["get"], detail=True, url_path="download-cover")
    def download_cover(self, request, slug=None):
        post = self.get_object()

        if not post.cover_image:
            raise Http404("Cover image not found")

        # as_attachment=True -> browser yuklab oladi
        return FileResponse(
            post.cover_image.open("rb"),
            as_attachment=True,
            filename=post.cover_image.name.split("/")[-1],
        )

class AuthApiTests(APITestCase):
    def test_register(self):

        res = self.client.post(url, payload, format="json")

        self.assertEqual(res.status_code, status.HTTP_200_OK)

from apps.common.tests.factories import create_category, create_post, create_tag


class PostApiTests(APITestCase):
    def test_posts_list_is_paginated(self):

            create_post(title=f"T{i}", slug=f"t{i}", category=cat)

        res = self.client.get("/api/v1/posts/")

        self.assertEqual(res.status_code, status.HTTP_200_OK)
        self.assertIn("results", res.data)
        self.assertIn("count", res.data)

        post = create_post(slug="my-post")

        res = self.client.get(f"/api/v1/posts/{post.slug}/")

        self.assertEqual(res.status_code, status.HTTP_200_OK)

        slugs = [p["slug"] for p in res.data["results"]]

        self.assertNotIn("p2", slugs)

        res = self.client.get(f"/api/v1/posts/?tags={tag.id}")

II BOB.  FASTAPI
Mazkur bobda FastAPI freymvorki asosida zamonaviy, yuqori unumdorlikka ega va asinxron backend ilovalarini ishlab chiqish jarayoni bosqichma-bosqich yoritiladi. Bob FastAPI loyihasini yaratishdan boshlanib, loyiha arxitekturasini to‘g‘ri tashkil etish, ma’lumotlar bazasini (MB) loyihalash va yaratish, muhit sozlamalarini amalga oshirish kabi dastlabki bosqichlarni qamrab oladi. ASGI konsepsiyasi, async/await mexanizmi va FastAPI ning ishlash prinsiplari amaliy misollar asosida tushuntiriladi.
Keyingi bosqichda foydalanuvchilarni boshqarish jarayoni ko‘rib chiqilib, identifikatsiya, autentifikatsiya va avtorizatsiya mexanizmlari chuqur tahlil qilinadi. Token, session va JWT asosidagi xavfsizlik yondashuvlari, parollarni xeshlash, foydalanuvchi rollari va ruxsat darajalarini boshqarish masalalari real loyiha misolida yoritiladi.
Bobda CRUD (Create, Read, Update, Delete) amallarini bajarish, model va schema (forma) tuzish, Pydantic yordamida ma’lumotlarni validatsiya qilish jarayonlari bosqichma-bosqich tushuntiriladi. Endpointlar yaratish, HTTP metodlari bilan ishlash va REST arxitektura tamoyillariga mos API ishlab chiqish ko‘nikmalari shakllantiriladi.
Shuningdek, ma’lumotlarni filtrlash, qidiruv (search) va tartiblash (ordering) mexanizmlari, katta hajmdagi ma’lumotlar bilan ishlash strategiyalari amaliy mashg‘ulotlar orqali mustahkamlanadi. Fayllarni yuklash va qayta ishlash, media ma’lumotlar bilan ishlash jarayonlari ham ko‘rib chiqilib, real loyihalarda qo‘llash usullari o‘rgatiladi.
Bobning yakuniy qismida tayyor loyihani Docker yordamida konteynerlash va deploy qilish jarayoni tushuntiriladi. Dockerfile yozish, muhit o‘zgaruvchilari bilan ishlash, konteyner ichida FastAPI ilovasini ishga tushirish bosqichlari amaliy tarzda namoyish etiladi. Shuningdek, FastAPI asosida real vaqt rejimida ishlovchi ilova yaratish — WebSocket texnologiyasi orqali ikki tomonlama aloqa o‘rnatish, real-time xabar almashish va interaktiv servislar ishlab chiqish masalalari yoritiladi.
Bob yakunida o‘quvchilar FastAPI asosida to‘liq funksional backend tizimini yaratish, xavfsiz API ishlab chiqish, uni konteynerlash va real vaqt rejimida ishlovchi servislarni ishlab chiqish ko‘nikmalariga ega bo‘ladilar hamda o‘rganilgan bilimlarni mustahkamlash maqsadida amaliy loyiha va nazorat topshiriqlari bilan bilimlarini sinovdan o‘tkazadilar.

Endi loyihani Identifikatsiya → Autentifikatsiya → Avtorizatsiya zanjiri bo‘yicha “toza” qilib davom ettiramiz. Biz FastAPI’da JWT (stateless token) modelini tanlaymiz (session ham tushuntirib o‘tiladi), va minimal, lekin production-ga yaqin arxitekturada qilamiz.
Quyidagilar bosqichma-bosqich amalga oshiriladi: model → migratsiya → hash → JWT → register/login → current user → role-based access.
Qisqacha nazariya
Identifikatsiya (Identification)
Foydalanuvchini “kim”ligini aytishi: masalan email yoki username.
Autentifikatsiya (Authentication)
U “haqiqatan o‘sha odammi?” tekshirish: password → hashed_password bilan solishtirish.
Avtorizatsiya (Authorization)
“U nima qila oladi?” tekshirish: masalan is_admin, yoki “owner” (o‘z resursiga egalik).
Token vs Session vs JWT
	•	Session (stateful): serverda session storage (Redis/DB) bo‘ladi, brauzer cookie’da session_id yuboradi.
	•	Token/JWT (stateless): server session saqlamaydi; tokenning o‘zida claimlar (sub, exp) bor. Har requestda Authorization: Bearer <token>.
Biz ushbu loyihamizda JWT ishlatamiz.

from datetime import datetime, timedelta, timezone
from typing import Any

from jose import jwt
from passlib.context import CryptContext

from app.core.config import settings

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")


def hash_password(password: str) -> str:
    return pwd_context.hash(password)


def verify_password(password: str, hashed_password: str) -> bool:
    return pwd_context.verify(password, hashed_password)


def create_access_token(subject: str, expires_minutes: int | None = None) -> str:
    expire = datetime.now(timezone.utc) + timedelta(
        minutes=expires_minutes or settings.access_token_expire_minutes
    )
    to_encode: dict[str, Any] = {"sub": subject, "exp": expire}
    return jwt.encode(to_encode, settings.jwt_secret_key, algorithm=settings.jwt_algorithm)

5.1. app/services/auth_service.py
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from app.core.security import hash_password, verify_password
from app.models.user import User


class AuthService:
    @staticmethod
    async def get_user_by_email(db: AsyncSession, email: str) -> User | None:
        result = await db.execute(select(User).where(User.email == email))
        return result.scalar_one_or_none()

    @staticmethod
    async def register(db: AsyncSession, full_name: str, email: str, password: str) -> User:
        existing = await AuthService.get_user_by_email(db, email)
        if existing:
            raise ValueError("EMAIL_ALREADY_EXISTS")

        user = User(
            full_name=full_name,
            email=email,
            hashed_password=hash_password(password),
            is_active=True,
            is_admin=False,
        )
        db.add(user)
        await db.commit()
        await db.refresh(user)
        return user

    @staticmethod
    async def authenticate(db: AsyncSession, email: str, password: str) -> User | None:
        user = await AuthService.get_user_by_email(db, email)
        if not user:
            return None
        if not user.is_active:
            return None
        if not verify_password(password, user.hashed_password):
            return None
        return user
ValueError("EMAIL_ALREADY_EXISTS") — keyin HTTP layer’da 400/409 ga map qilamiz.

6.1. app/api/deps.py
from fastapi import Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from jose import JWTError, jwt
from sqlalchemy.ext.asyncio import AsyncSession

from app.core.config import settings
from app.core.database import get_db
from app.models.user import User
from app.services.auth_service import AuthService

oauth2_scheme = OAuth2PasswordBearer(tokenUrl="/auth/login")


async def get_current_user(
    token: str = Depends(oauth2_scheme),
    db: AsyncSession = Depends(get_db),
) -> User:
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Could not validate credentials",
        headers={"WWW-Authenticate": "Bearer"},
    )

    try:
        payload = jwt.decode(token, settings.jwt_secret_key, algorithms=[settings.jwt_algorithm])
        email: str | None = payload.get("sub")
        if not email:
            raise credentials_exception
    except JWTError:
        raise credentials_exception

    user = await AuthService.get_user_by_email(db, email)
    if not user or not user.is_active:
        raise credentials_exception
    return user


def require_admin(user: User = Depends(get_current_user)) -> User:
    if not user.is_admin:
        raise HTTPException(status_code=403, detail="Admin privileges required")
    return user
Bu yerda avtorizatsiya boshlandi: require_admin — role check.

from fastapi import APIRouter, Depends, HTTPException, status
from fastapi.security import OAuth2PasswordRequestForm
from sqlalchemy.ext.asyncio import AsyncSession

from app.api.deps import get_current_user
from app.core.database import get_db
from app.core.security import create_access_token
from app.models.user import User
from app.schemas.token import Token
from app.schemas.user import UserCreate, UserPublic
from app.services.auth_service import AuthService

router = APIRouter(prefix="/auth", tags=["auth"])


@router.post("/register", response_model=UserPublic, status_code=status.HTTP_201_CREATED)
async def register(payload: UserCreate, db: AsyncSession = Depends(get_db)) -> User:
    try:
        user = await AuthService.register(
            db=db,
            full_name=payload.full_name,
            email=str(payload.email),
            password=payload.password,
        )
        return user
    except ValueError as e:
        if str(e) == "EMAIL_ALREADY_EXISTS":
            raise HTTPException(status_code=409, detail="Email already registered")
        raise


@router.post("/login", response_model=Token)
async def login(
    form: OAuth2PasswordRequestForm = Depends(),
    db: AsyncSession = Depends(get_db),
) -> Token:
    user = await AuthService.authenticate(db=db, email=form.username, password=form.password)
    if not user:
        raise HTTPException(status_code=401, detail="Incorrect email or password")

    access_token = create_access_token(subject=user.email)
    return Token(access_token=access_token)


@router.get("/me", response_model=UserPublic)
async def me(current_user: User = Depends(get_current_user)) -> User:
    return current_user

import uuid

from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from app.models.book import Book
from app.models.category import Category
from app.models.user import User


class BookService:
    @staticmethod
    async def list_public(db: AsyncSession, limit: int = 20, offset: int = 0) -> list[Book]:
        stmt = (
            select(Book)
            .where(Book.is_published.is_(True))
            .order_by(Book.created_at.desc())
            .limit(limit)
            .offset(offset)
        )
        result = await db.execute(stmt)
        return list(result.scalars().all())

    @staticmethod
    async def get_public(db: AsyncSession, book_id: uuid.UUID) -> Book | None:
        stmt = select(Book).where(Book.id == book_id, Book.is_published.is_(True))
        result = await db.execute(stmt)
        return result.scalar_one_or_none()

    @staticmethod
    async def get_any(db: AsyncSession, book_id: uuid.UUID) -> Book | None:
        result = await db.execute(select(Book).where(Book.id == book_id))
        return result.scalar_one_or_none()

    @staticmethod
    async def create(
        db: AsyncSession,
        owner: User,
        *,
        title: str,
        description: str | None,
        author: str | None,
        language: str | None,
        published_year: int | None,
        pages: int | None,
        category_id: uuid.UUID,
        is_published: bool,
    ) -> Book:
        # Biznes validatsiya: category mavjud bo‘lishi shart
        cat = await db.execute(select(Category).where(Category.id == category_id))
        if not cat.scalar_one_or_none():
            raise ValueError("CATEGORY_NOT_FOUND")

        book = Book(
            title=title,
            description=description,
            author=author,
            language=language,
            published_year=published_year,
            pages=pages,
            category_id=category_id,
            owner_id=owner.id,
            is_published=is_published,
            file_path=None,
            cover_path=None,
        )
        db.add(book)
        await db.commit()
        await db.refresh(book)
        return book

    @staticmethod
    def ensure_owner_or_admin(book: Book, user: User) -> None:
        if user.is_admin:
            return
        if book.owner_id != user.id:
            raise PermissionError("NOT_OWNER")

    @staticmethod
    async def update(
        db: AsyncSession,
        book: Book,
        user: User,
        *,
        title: str | None,
        description: str | None,
        author: str | None,
        language: str | None,
        published_year: int | None,
        pages: int | None,
        category_id: uuid.UUID | None,
        is_published: bool | None,
    ) -> Book:
        BookService.ensure_owner_or_admin(book, user)

        if category_id is not None:
            cat = await db.execute(select(Category).where(Category.id == category_id))
            if not cat.scalar_one_or_none():
                raise ValueError("CATEGORY_NOT_FOUND")
            book.category_id = category_id

        if title is not None:
            book.title = title
        if description is not None:
            book.description = description
        if author is not None:
            book.author = author
        if language is not None:
            book.language = language
        if published_year is not None:
            book.published_year = published_year
        if pages is not None:
            book.pages = pages
        if is_published is not None:
            book.is_published = is_published

        await db.commit()
        await db.refresh(book)
        return book

    @staticmethod
    async def delete(db: AsyncSession, book: Book, user: User) -> None:
        BookService.ensure_owner_or_admin(book, user)
        await db.delete(book)
        await db.commit()

import uuid

from sqlalchemy import and_, desc, or_, select
from sqlalchemy.ext.asyncio import AsyncSession

from app.models.book import Book
from app.schemas.book_query import SortBy, SortOrder


class BookService:
    @staticmethod
    async def list_public(
        db: AsyncSession,
        *,
        q: str | None = None,
        category_id: uuid.UUID | None = None,
        author: str | None = None,
        language: str | None = None,
        year_from: int | None = None,
        year_to: int | None = None,
        pages_from: int | None = None,
        pages_to: int | None = None,
        sort_by: SortBy = SortBy.created_at,
        sort_order: SortOrder = SortOrder.desc,
        limit: int = 20,
        offset: int = 0,
    ) -> list[Book]:
        conditions = [Book.is_published.is_(True)]

        # --- filters ---
        if category_id is not None:
            conditions.append(Book.category_id == category_id)

        if language is not None:
            conditions.append(Book.language == language)

        if year_from is not None:
            conditions.append(Book.published_year >= year_from)
        if year_to is not None:
            conditions.append(Book.published_year <= year_to)

        if pages_from is not None:
            conditions.append(Book.pages >= pages_from)
        if pages_to is not None:
            conditions.append(Book.pages <= pages_to)

        # --- search ---
        # q: title/author bo‘yicha ILIKE
        if q:
            pattern = f"%{q.strip()}%"
            conditions.append(or_(Book.title.ilike(pattern), Book.author.ilike(pattern)))

        # author filter (alohida) - partial qidiruv sifatida
        if author:
            pattern = f"%{author.strip()}%"
            conditions.append(Book.author.ilike(pattern))

        stmt = select(Book).where(and_(*conditions))

        # --- sorting (allowlist) ---
        sort_map = {
            SortBy.created_at: Book.created_at,
            SortBy.title: Book.title,
            SortBy.published_year: Book.published_year,
            SortBy.pages: Book.pages,
        }
        sort_col = sort_map[sort_by]
        if sort_order == SortOrder.desc:
            stmt = stmt.order_by(desc(sort_col))
        else:
            stmt = stmt.order_by(sort_col)

        stmt = stmt.limit(limit).offset(offset)

        result = await db.execute(stmt)
        return list(result.scalars().all())

	•	and_(*conditions): hammasini bir joyda WHERE ga yig‘adi.

            raise HTTPException(status_code=422, detail="Invalid category_id")

    # Kichik biznes validatsiya:

        raise HTTPException(status_code=422, detail="year_from cannot be greater than year_to")
    if pages_from is not None and pages_to is not None and pages_from > pages_to:
        raise HTTPException(status_code=422, detail="pages_from cannot be greater than pages_to")

    return await BookService.list_public(
        db,
        q=q,
        category_id=cid,
        author=author,
        language=language,
        year_from=year_from,
        year_to=year_to,
        pages_from=pages_from,
        pages_to=pages_to,
        sort_by=sort_by,
        sort_order=sort_order,
        limit=limit,
        offset=offset,

from app.models.user import User

    @staticmethod
    async def list_my(
        db: AsyncSession,
        *,
        user: User,
        q: str | None = None,
        is_published: bool | None = None,
        sort_by: SortBy = SortBy.created_at,
        sort_order: SortOrder = SortOrder.desc,
        limit: int = 20,
        offset: int = 0,
    ) -> list[Book]:
        conditions = [Book.owner_id == user.id]

        if is_published is not None:
            conditions.append(Book.is_published.is_(is_published))

        if q:
            pattern = f"%{q.strip()}%"
            conditions.append(or_(Book.title.ilike(pattern), Book.author.ilike(pattern)))

        stmt = select(Book).where(and_(*conditions))

        sort_map = {
            SortBy.created_at: Book.created_at,
            SortBy.title: Book.title,
            SortBy.published_year: Book.published_year,
            SortBy.pages: Book.pages,
        }
        sort_col = sort_map[sort_by]
        stmt = stmt.order_by(desc(sort_col) if sort_order == SortOrder.desc else sort_col)

        stmt = stmt.limit(limit).offset(offset)
        result = await db.execute(stmt)
        return list(result.scalars().all())

import os
import uuid
from pathlib import Path
from typing import Iterable

from fastapi import UploadFile

from app.core.config import settings


class StorageService:
    def __init__(self) -> None:
        self.media_root = Path(settings.media_root)

    def ensure_dirs(self) -> None:
        (self.media_root / "books").mkdir(parents=True, exist_ok=True)
        (self.media_root / "covers").mkdir(parents=True, exist_ok=True)

    @staticmethod
    def _safe_ext(filename: str) -> str:
        ext = Path(filename).suffix.lower().strip(".")
        return ext

    @staticmethod
    def _validate_ext(ext: str, allowed: Iterable[str]) -> None:
        if ext not in allowed:
            raise ValueError("INVALID_EXTENSION")

    @staticmethod
    def _validate_size(size_bytes: int, max_bytes: int) -> None:
        if size_bytes > max_bytes:
            raise ValueError("FILE_TOO_LARGE")

    @staticmethod
    async def _get_size(upload: UploadFile) -> int:
        # UploadFile.file - SpooledTemporaryFile
        fileobj = upload.file
        current = fileobj.tell()
        fileobj.seek(0, os.SEEK_END)
        size = fileobj.tell()
        fileobj.seek(current, os.SEEK_SET)
        return size

    async def save_book_file(self, upload: UploadFile) -> str:
        """
        Accepts: pdf, epub
        Returns: relative path like 'media/books/<uuid>.<ext>'
        """
        self.ensure_dirs()

        ext = self._safe_ext(upload.filename or "")
        self._validate_ext(ext, allowed=("pdf", "epub"))

        # MIME type tekshirish (qisman ishonchli, lekin foydali)
        allowed_types = {"application/pdf", "application/epub+zip"}
        if upload.content_type not in allowed_types:
            raise ValueError("INVALID_CONTENT_TYPE")

        size = await self._get_size(upload)
        self._validate_size(size, max_bytes=20 * 1024 * 1024)  # 20MB

        filename = f"{uuid.uuid4()}.{ext}"
        relative = Path(settings.media_root) / "books" / filename
        absolute = Path.cwd() / relative

        # Diskka yozish (stream)
        with absolute.open("wb") as f:
            while chunk := await upload.read(1024 * 1024):  # 1MB chunk
                f.write(chunk)

        await upload.close()
        return str(relative)

    async def save_cover_file(self, upload: UploadFile) -> str:
        """
        Accepts: jpg, jpeg, png, webp
        Returns: relative path like 'media/covers/<uuid>.<ext>'
        """
        self.ensure_dirs()

        ext = self._safe_ext(upload.filename or "")
        self._validate_ext(ext, allowed=("jpg", "jpeg", "png", "webp"))

        allowed_types = {"image/jpeg", "image/png", "image/webp"}
        if upload.content_type not in allowed_types:
            raise ValueError("INVALID_CONTENT_TYPE")

        size = await self._get_size(upload)
        self._validate_size(size, max_bytes=5 * 1024 * 1024)  # 5MB

        filename = f"{uuid.uuid4()}.{ext}"
        relative = Path(settings.media_root) / "covers" / filename
        absolute = Path.cwd() / relative

        with absolute.open("wb") as f:
            while chunk := await upload.read(1024 * 1024):
                f.write(chunk)

        await upload.close()
        return str(relative)

    def delete_file(self, relative_path: str | None) -> None:
        if not relative_path:
            return
        absolute = Path.cwd() / relative_path
        if absolute.exists() and absolute.is_file():
            absolute.unlink()

            conditions.append(Book.pages >= pages_from)
        if pages_to is not None:
            conditions.append(Book.pages <= pages_to)

        if q:
            pattern = f"%{q.strip()}%"
            conditions.append(or_(Book.title.ilike(pattern), Book.author.ilike(pattern)))

        if author:
            pattern = f"%{author.strip()}%"
            conditions.append(Book.author.ilike(pattern))

        where_clause = and_(*conditions)

        # total count
        total_stmt = select(func.count()).select_from(Book).where(where_clause)
        total = (await db.execute(total_stmt)).scalar_one()

        # items
        stmt = select(Book).where(where_clause)

        sort_map = {
            SortBy.created_at: Book.created_at,
            SortBy.title: Book.title,
            SortBy.published_year: Book.published_year,
            SortBy.pages: Book.pages,
        }
        sort_col = sort_map[sort_by]
        stmt = stmt.order_by(desc(sort_col) if sort_order == SortOrder.desc else sort_col)

        stmt = stmt.limit(limit).offset(offset)

        result = await db.execute(stmt)
        items = list(result.scalars().all())
        return total, items

async def get_book_private(
    book_id: str,
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    try:
        bid = uuid.UUID(book_id)
    except ValueError:
        raise HTTPException(status_code=422, detail="Invalid book_id")

    book = await BookService.get_any(db, bid)
    if not book:

        BookService.ensure_owner_or_admin(book, current_user)

from typing import List
from sqlalchemy.orm import Session
from sqlalchemy import select, desc

from app.domain.models import ChatMessage


class MessageRepository:
    def create(self, db: Session, room: str, username: str, content: str) -> ChatMessage:
        msg = ChatMessage(room=room, username=username, content=content)
        db.add(msg)
        db.commit()
        db.refresh(msg)
        return msg

    def list_last(self, db: Session, room: str, limit: int = 50) -> List[ChatMessage]:
        stmt = (
            select(ChatMessage)
            .where(ChatMessage.room == room)
            .order_by(desc(ChatMessage.created_at))
            .limit(limit)
        )
        rows = db.execute(stmt).scalars().all()
        # UI uchun eski->yangi qilib qaytaramiz
        return list(reversed(rows))
4.7. app/services/connection_manager.py
import asyncio
from collections import defaultdict
from typing import DefaultDict, Dict, Set

from fastapi import WebSocket


class ConnectionManager:
    """
    Room bo‘yicha WebSocket ulanishlarini boshqaradi.
    Thread-safe emas, lekin asyncio event loop ichida xavfsiz.
    """

    def __init__(self) -> None:
        self._rooms: DefaultDict[str, Set[WebSocket]] = defaultdict(set)
        self._lock = asyncio.Lock()

    async def connect(self, room: str, websocket: WebSocket) -> None:
        await websocket.accept()
        async with self._lock:
            self._rooms[room].add(websocket)

    async def disconnect(self, room: str, websocket: WebSocket) -> None:
        async with self._lock:
            if websocket in self._rooms[room]:
                self._rooms[room].remove(websocket)
            if not self._rooms[room]:
                self._rooms.pop(room, None)

    async def broadcast(self, room: str, message: Dict) -> None:
        """
        Room ichidagi barcha websocketlarga JSON yuboradi.
        Xatolik bo‘lsa, o‘sha websocket keyin disconnect qilinadi.
        """
        async with self._lock:
            sockets = list(self._rooms.get(room, set()))

        to_remove: list[WebSocket] = []
        for ws in sockets:
            try:
                await ws.send_json(message)
            except Exception:
                to_remove.append(ws)

        if to_remove:
            async with self._lock:
                for ws in to_remove:
                    if ws in self._rooms.get(room, set()):
                        self._rooms[room].remove(ws)

import logging
from fastapi import APIRouter, Depends, Query, WebSocket, WebSocketDisconnect, status
from sqlalchemy.orm import Session

from app.db.session import get_db
from app.domain.schemas import WsEnvelope
from app.repositories.messages import MessageRepository
from app.services.chat_service import ChatService
from app.services.connection_manager import ConnectionManager

router = APIRouter()
logger = logging.getLogger("ws")

manager = ConnectionManager()
chat_service = ChatService(repo=MessageRepository())


@router.websocket("/ws/chat/{room}")
async def websocket_chat(
    websocket: WebSocket,
    room: str,
    username: str = Query(default="anonymous", min_length=1, max_length=50),
    db: Session = Depends(get_db),
) -> None:
    # 1) Ulanish
    await manager.connect(room=room, websocket=websocket)

    join_event = WsEnvelope(
        type="user:join",
        room=room,
        username=username,
        payload={"message": f"{username} joined"},
    )
    await manager.broadcast(room=room, message=join_event.model_dump())

    try:
        # 2) Loop: kelayotgan xabarlarni qabul qilish
        while True:
            data = await websocket.receive_json()

            # Minimal protokol: {"content": "..."}
            content = (data.get("content") or "").strip()
            if not content:
                err = WsEnvelope(
                    type="error",
                    room=room,
                    username=username,
                    payload={"detail": "Empty message is not allowed"},
                )
                await websocket.send_json(err.model_dump())
                continue

            event = chat_service.save_message(
                db=db,
                room=room,
                username=username,
                content=content,
            )
            await manager.broadcast(room=room, message=event.model_dump())

    except WebSocketDisconnect:
        logger.info("Client disconnected: room=%s username=%s", room, username)
    except Exception as exc:
        logger.exception("WebSocket error: %s", exc)
        await websocket.close(code=status.WS_1011_INTERNAL_ERROR)
    finally:
        await manager.disconnect(room=room, websocket=websocket)
        left_event = WsEnvelope(
            type="user:left",
            room=room,
            username=username,
            payload={"message": f"{username} left"},
        )
        await manager.broadcast(room=room, message=left_event.model_dump())

	•	broadcast() — room bo‘yicha tarqatish
B) Nega DB (repository) ajratdik?
DB logikani WebSocket handler ichida qoldirsak kod “shishib” ketadi. Repository:
	•	create() — xabarni saqlaydi
	•	list_last() — oxirgi N ta xabarni oladi
Service (ChatService) esa biznes oqimini “oraliq qatlam” sifatida boshqaradi.
C) WebSocket protokol (oddiy envelope)
Server har doim bir xil formatda jo‘natadi:
	•	type: event nomi
	•	room: qaysi room
	•	username: kim yuborgan
	•	payload: eventga mos data
Bu keyinchalik “typing indicator”, “read receipts” kabi funksiyalarni qo‘shishni osonlashtiradi.
D) History nima uchun HTTP endpoint orqali?
WebSocket faqat real-time. Sahifa ochilganda avval history ko‘rsatish uchun:
	•	GET /rooms/{room}/messages?limit=50
Keyin WebSocket real-time davom etadi.

	•	Redis pub/sub (multi-worker / scale uchun)
	•	Rate limiting (floodga qarshi)

III BOB. PROMPT ENGINEERING
Mazkur bobda Prompt Engineering asoslari va katta til modellari (LLM) bilan samarali ishlash metodologiyasi bosqichma-bosqich yoritiladi. Bob rollar va kontekst asosidagi promptlar tuzishdan boshlanib, modelga beriladigan topshiriqni aniq va strukturalangan shaklda ifodalash tamoyillarini shakllantirishga yo‘naltirilgan. System, user va assistant rollarining vazifalari, kontekst boshqaruvi hamda model javobiga ta’sir qiluvchi omillar amaliy misollar orqali tushuntiriladi.
Shuningdek, Chain-of-Thought (fikrlash zanjiri) yondashuvi asosida murakkab masalalarni bosqichma-bosqich yechish, Zero-shot, One-shot va Few-shot prompting strategiyalari yordamida modelni qo‘shimcha misollar orqali yo‘naltirish usullari o‘rgatiladi. ReAct prompting va reasoning mexanizmlari orqali modelning tahliliy fikrlash qobiliyatini oshirish, qaror qabul qilish jarayonlarini strukturalash ko‘nikmalari hosil qilinadi.
Bobda kod yozish va ma’lumotlar bilan ishlash uchun promptlar tuzish, strukturaviy javoblar olish, JSON yoki kod formatida natija generatsiya qilish usullari ko‘rib chiqiladi. Matn tahlili, hujjatlar bilan ishlash, mazmun ajratish, xulosa chiqarish va ma’lumotlarni transformatsiya qilish vazifalari amaliy mashg‘ulotlar orqali mustahkamlanadi. Shuningdek, rasm va dizayn promptlari, multimodal yondashuvlar, ovozli va vizual ma’lumotlar bilan ishlash imkoniyatlari yoritiladi.
Bob davomida domen-spetsifik yordamchi yoki Custom GPT yaratish, bilimlar bazasi bilan integratsiya qilish, aniq sohaga moslashtirilgan AI yordamchi ishlab chiqish jarayonlari bosqichma-bosqich tushuntiriladi. Etika, xavfsizlik, faktlarni tekshirish (fact-checking) va noto‘g‘ri ma’lumotlar xavfini kamaytirish masalalariga alohida e’tibor qaratiladi. API orqali modelni backend tizimlarga ulash va avtomatlashtirilgan jarayonlar yaratish usullari ham ko‘rib chiqiladi.
Bob yakunida o‘quvchilar domen asosida mini loyiha ishlab chiqib, AI yordamchi yaratadilar hamda yakuniy loyiha taqdimoti orqali o‘z yechimlarini tahlil qiladilar. Chain-of-Thought (CoT) prompting – AI modelini to‘g‘ridan-to‘g‘ri javob berish o‘rniga, muammo ustida bosqichma-bosqich fikrlashga undash texnikasi. Bu ayniqsa murakkab mantiqiy masalalar, matematik hisob-kitoblar va ko‘p bosqichli tahlillarda juda samarali.
CoT asosiy tamoyili
Odamlar murakkab masalalarni ko‘pincha bosqichlarga bo‘lib hal qilishadi. Xuddi shunday, AI modeliga ham "fikrlash zanjiri" ko‘rsatilsa, u aniqroq va ishonchliroq javob beradi.
📌 Standart vs Chain-of-Thought prompt
❌ Oddiy: "Bir do‘konda 15 ta olma bor edi. 6 tasi sotildi, keyin yana 8 ta keldi. Nechta olma bor?" → AI yanglishishi mumkin

✅ CoT bilan: "Bir do‘konda 15 ta olma bor edi. 6 tasi sotildi, keyin yana 8 ta keldi. Nechta olma bor? Iltimos, har bir amalni alohida ko‘rsatib, qadamma-qadam hisoblang."

Prompt injection va Xavfsizlik
Prompt injection – zararli foydalanuvchilarning AI tizimiga noto‘g‘ri buyruqlar kiritib, xavfsizlik cheklovlarini chetlab o‘tishga urinishi. Bu ayniqsa public-facing AI tizimlarida muhim muammo.
⚠️ Xavfsizlik tamoyillari
• System promptda aniq cheklovlarni kiriting: 'Hech qachon...ni bajarmang‘
• Foydalanuvchi kiritgan matnni to‘g‘ridan-to‘g‘ri prompt ichiga kiritmang
• Chiqish filtrlari va content moderation qo‘llang
• Shaxsiy ma'lumotlarni promptlarga kiritmang (GDPR/PDPA)

Ishni bajarish tartibi
Prompt engineering‘ni to‘liq potensialini ochish uchun uni dasturiy ta'minot bilan integratsiya qilish zarur. Bu darsda OpenAI API, LangChain va boshqa vositalar yordamida AI ni tizimlar ichiga qanday kiritishni o‘rganamiz.
OpenAI API asoslari
Python: OpenAI API bilan ishlash
import openai
from openai import OpenAI
 
client = OpenAI(api_key="your-api-key")
 
def prompt_with_system(system_prompt, user_prompt, model="gpt-4o"):
    response = client.chat.completions.create(
        model=model,
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": user_prompt}
        ],
        temperature=0.7,      # Kreativlik darajasi (0–1)
        max_tokens=1000,      # Maksimal javob uzunligi
        top_p=0.9,            # Nucleus sampling
    )
    return response.choices[0].message.content
 
# Misol:
system = "Sen O‘zbekiston tarixi bo‘yicha ekspertsan."
user = "Amir Temur davrining eng muhim yutuqlari nima?"
print(prompt_with_system(system, user))
LangChain bilan prompt zanjiri
from langchain.chat_models import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.chains import LLMChain
 
# Prompt template yaratish
template = ChatPromptTemplate.from_messages([
    ("system", "Sen {soha} bo‘yicha mutaxassissan."),
    ("human", "{savol}")
])
 
# Chain yaratish
llm = ChatOpenAI(model="gpt-4o", temperature=0)
chain = LLMChain(llm=llm, prompt=template)
 
# Ishlatish
natija = chain.run(soha="marketing", savol="Ijtimoiy tarmoqlar uchun eng yaxshi kontent strategiyasi?")
print(natija)

Ishni bajarish tartibi

Loyiha topshirig‘i
📋 Mini loyiha talablari
1. Domen tanlash: Ta'lim, tibbiyot, huquq, savdo, texnologiya va h.k.
2. System prompt: Kamida 200 so‘zlik batafsil tizim ko‘rsatmasi
3. Prompt kutubxonasi: Domenga xos 10 ta tayyor prompt
4. Xavfsizlik: Cheklovlar va etik qoidalar ro‘yxati
5. Test: Kamida 5 ta test suhbat va natijalar tahlili
Loyiha bosqichlari
	•	Domen tadqiqoti: Soha talablari, foydalanuvchilar, umumiy savollar
	•	System prompt draft: Birinchi versiya yozish
	•	Iterativ yaxshilash: Test → Tahlil → Tuzatish tsikli (kamida 3 marta)
	•	Prompt kutubxonasi yaratish: Tez-tez ishlatiladigan promptlar
	•	Dokumentatsiya: Foydalanish ko‘rsatmasi va cheklovlar

💡 Muvaffaqiyatli Mini Loyiha Misoli: O‘zbek Til O‘qituvchisi
Domen: O‘zbek tili ta'limi
Maqsad auditoriya: O‘zbek tilini o‘rganayotgan chet elliklar
Asosiy funksiyalar: Grammatika izohi, lug‘at, tarjima, amaliy mashqlar
Uslub: Sabr-toqatli, rag‘batlantiruvchi, amaliylashtirilgan
Cheklovlar: Hech qachon noto‘g‘ri grammatika namuna keltirmaydi,
